Spells will be balanced following these rules
The most optimal damage spell will do 35 damage and 5 resource points. If it’s less than 35 it can cost less resources. Same for healing spells 35 is the optimal heal and 5 is resource cost
If the damage spell is more than 35 it must be balanced by costing more resources and with additional penalties such as self damage, self status or debuff, Low accuracy etc. 
Spells with secondary status effects should do less damage and cost more. 
Spell range should depend on the type of attack. Bow, Gun or Magic should have 2-3 grid tiles range . Melee should always be limited to adjacent grids, so range should be 1 for melee unless it’s a weapon throw or spear then it can be 2 but never more than 2. 
If the spell does more than 70 damage or heals they should cause a catastrophic side effect like a 5% chance Spell casting NPC will die.
All support type moves except Healing will be highly unpredictable with extreme cases of 1% accuracy or 50% chance of causing self harm. 
All scripts that are generated will have a full working function not just only a debug.

"using UnityEngine;
 using System.Collections.Generic;

public class TurnBasedSummonExample : MonoBehaviour
{
    public GameObject summonerPrefab;
    public GameObject monsterPrefab;

    private List<Unit> allUnits = new List<Unit>();
    private int currentUnitIndex = 0;

    private void Start()
    {
        Unit summoner = SpawnUnit(summonerPrefab, new Vector2Int(2, 1), Team.Player);
        Unit enemy = SpawnUnit(summonerPrefab, new Vector2Int(2, 4), Team.Enemy);
        allUnits.Add(summoner);
        allUnits.Add(enemy);
        StartTurn();
    }

    private Unit SpawnUnit(GameObject prefab, Vector2Int gridPos, Team team)
    {
        Vector3 pos = new Vector3(gridPos.x, 0, gridPos.y);
        GameObject obj = Instantiate(prefab, pos, Quaternion.identity);
        Unit u = obj.AddComponent<Unit>();
        u.gridPos = gridPos;
        u.team = team;
        u.example = this;
        return u;
    }

    private void StartTurn()
    {
        if (allUnits.Count == 0) return;
        Unit current = allUnits[currentUnitIndex];
        current.OnTurnStart();
    }

    public void EndTurn(Unit unit)
    {
        currentUnitIndex = (currentUnitIndex + 1) % allUnits.Count;
        StartTurn();
    }

    public void RegisterSummon(Unit summon)
    {
        allUnits.Insert(currentUnitIndex + 1, summon);
    }
}
